\begin{exercise}{Konjugerede komplekse tal 1-1}

Bestem det konjugerede af det komplekse tal $3 + 4i$.

\answerbox{3 - 4i}

\hint

Det konjugerede af et komplekst tal $a + bi$ er $a - bi$.

\hint

For $3 + 4i$:

\hint
\[
\overline{3 + 4i} = 3 - 4i
\]

\end{exercise}

\hint

\overline{-2 + 5i} = -2 - 5i
\]

\end{exercise}

\begin{exercise}{Konjugerede komplekse tal 1-3}

Bestem det konjugerede af det komplekse tal $7i$.

\answerbox{-7i}

\hint

Skriv som $0 + 7i$:

\hint
\[
7i = 0 + 7i
\]

\hint

Det konjugerede:

\hint
\[
\overline{0 + 7i} = 0 - 7i = -7i
\]

\end{exercise}

\begin{exercise}{Konjugerede komplekse tal 1-4}

Bestem det konjugerede af det komplekse tal $5$.

\answerbox{5}

\hint

Reelle tal har ingen imaginær del:

\hint
\[
5 = 5 + 0i
\]

\hint

Det konjugerede:

\hint
\[
\overline{5 + 0i} = 5 - 0i = 5
\]

\end{exercise}

\begin{exercise}{Konjugerede komplekse tal 1-6}

Bestem det konjugerede af det komplekse tal $a + bi$, hvor $a$ og $b$ er reelle tal.

\answerbox{a - bi}

\hint

Generel regel for det konjugerede af et komplekst tal:

\hint
\[
\overline{a + bi} = a - bi
\]
