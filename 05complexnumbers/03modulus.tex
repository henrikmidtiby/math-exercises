\begin{exercise}{Modulus af komplekse tal 1-1}

Bestem modulus af det komplekse tal $3 + 4i$.

\answerbox{5}

\hint

Modulus af et komplekst tal $a + bi$ er $\sqrt{a^2 + b^2}$.

\hint

For $3 + 4i$:

\hint
\[
|3 + 4i| = \sqrt{3^2 + 4^2} = \sqrt{9 + 16} = \sqrt{25} = 5
\]

\end{exercise}

\begin{exercise}{Modulus af komplekse tal 1-2}

Bestem modulus af det komplekse tal $1 + i$.

\answerbox{\sqrt{2}}

\hint

Anvend modulusformlen:

\hint
\[
|1 + i| = \sqrt{1^2 + 1^2} = \sqrt{1 + 1} = \sqrt{2}
\]

\end{exercise}

\begin{exercise}{Modulus af komplekse tal 1-3}

Bestem modulus af det komplekse tal $5 - 12i$.

\answerbox{13}

\hint

Anvend modulusformlen:

\hint
\[
|5 - 12i| = \sqrt{5^2 + (-12)^2} = \sqrt{25 + 144} = \sqrt{169} = 13
\]

\end{exercise}

\begin{exercise}{Modulus af komplekse tal 1-4}

Bestem modulus af det komplekse tal $7i$.

\answerbox{7}

\hint

Skriv som $0 + 7i$:

\hint
\[
|0 + 7i| = \sqrt{0^2 + 7^2} = \sqrt{49} = 7
\]

\end{exercise}

\begin{exercise}{Modulus af komplekse tal 1-5}

Bestem modulus af det komplekse tal $-6$.

\answerbox{6}

\hint

Reelle tal har modulus lig med absolutværdi:

\hint
\[
|-6| = \sqrt{(-6)^2 + 0^2} = \sqrt{36} = 6
\]

\end{exercise}

\begin{exercise}{Modulus af komplekse tal 1-6}

Bestem modulus af det komplekse tal $a + bi$, hvor $a = 3$ og $b = -4$.

\answerbox{5}

\hint

Indsæt værdierne i modulusformlen:

\hint
\[
|3 + (-4)i| = |3 - 4i| = \sqrt{3^2 + (-4)^2} = \sqrt{9 + 16} = \sqrt{25} = 5
\]

\end{exercise}
