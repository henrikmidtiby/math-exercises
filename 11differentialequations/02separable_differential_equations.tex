\begin{exercise}{Separable differentialligninger 1-1}

Løs den differentialligning $\frac{dy}{dx} = 2xy$.

\answerbox{y = Ce^{x^2}}

\hint

Adskil variablene: $\frac{dy}{y} = 2x \, dx$.

\hint

Integrer begge sider:
\[
\int \frac{1}{y} \, dy = \int 2x \, dx
\]

\hint

Få:
\[
\ln|y| = x^2 + C
\]

\hint

Eksponentier begge sider:
\[
y = e^{x^2 + C} = e^C \cdot e^{x^2} = Ce^{x^2}
\]

\end{exercise}

\begin{exercise}{Separable differentialligninger 1-2}

Løs den differentialligning $\frac{dy}{dx} = \frac{x}{y}$.

\answerbox{y = \sqrt{C + x^2}}

\hint

Adskil variablene:
\[
y \, dy = x \, dx
\]

\hint

Integrer begge sider:
\[
\int y \, dy = \int x \, dx
\]

\hint

Få:
\[
\frac{y^2}{2} = \frac{x^2}{2} + C
\]

\hint

Multiplicer med 2:
\[
y^2 = x^2 + 2C
\]

\hint

Skriv 2C som en ny konstant:
\[
y^2 = x^2 + C \\ y = \sqrt{C + x^2}
\]

\end{exercise}

\begin{exercise}{Separable differentialligninger 1-4}

Løs den differentialligning $\frac{dy}{dx} = y^2 x$.

\answerbox{y = -\frac{1}{C + \frac{x^2}{2}}}

\hint

Adskil variablene:
\[
\frac{dy}{y^2} = x \, dx
\]

\hint

Integrer begge sider:
\[
\int y^{-2} \, dy = \int x \, dx
\]

\hint

Få:
\[
-\frac{1}{y} = \frac{x^2}{2} + C
\]

\hint

Løs for y:
\[
y = -\frac{1}{C + \frac{x^2}{2}}
\]

\end{exercise}

\begin{exercise}{Separable differentialligninger 1-5}

Løs den differentialligning $\frac{dy}{dx} = \frac{y}{x}$ med begyndelsesbetingelsen $y(1) = 2$.

\answerbox{y = 2x}

\hint

Adskil variablene:
\[
\frac{dy}{y} = \frac{dx}{x}
\]

\hint

Integrer begge sider:
\[
\ln|y| = \ln|x| + C
\]

\hint

Eksponentier begge sider:
\[
y = e^C \cdot x = Cx
\]

\hint

Brug begyndelsesbetingelsen $y(1) = 2$:
\[
2 = C \cdot 1 \\ C = 2
\]

\hint

Løsning:
\[
y = 2x
\]

\end{exercise}

\begin{exercise}{Separable differentialligninger 1-6}

Løs den differentialligning $\frac{dy}{dx} = xy^2$ med begyndelsesbetingelsen $y(0) = 1$.

\answerbox{y = \frac{1}{1 - \frac{x^2}{2}}}

\hint

Adskil variablene:
\[
\frac{dy}{y^2} = x \, dx
\]

\hint

Integrer begge sider:
\[
\int y^{-2} \, dy = \int x \, dx
\]

\hint

Få:
\[
-\frac{1}{y} = \frac{x^2}{2} + C
\]

\hint

Brug begyndelsesbetingelsen $y(0) = 1$:
\[
-1 = 0 + C \\ C = -1
\]

\hint

Løs for y:
\[
y = -\frac{1}{-1 + \frac{x^2}{2}} = \frac{1}{1 - \frac{x^2}{2}}
\]

\end{exercise}
