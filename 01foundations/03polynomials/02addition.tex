\begin{exercise}{Addition af polynomier 1-1}

Bestem $(2x + 3) + (x - 5)$.

\answerbox{3x - 2}

\hint

Fjern parenteser (de er alle positive):

\hint
\[
2x + 3 + x - 5
\]

\hint

Saml ens led sammen:

\hint
\[
(2x + x) + (3 - 5)
\]

\hint

Udregn hver gruppes sum:

\hint
\[
3x - 2
\]

\end{exercise}

\begin{exercise}{Addition af polynomier 1-2}

Bestem $(3x^2 + 2x + 5) + (x^2 - 4x + 4)$.

\answerbox{4x^2 - 2x + 9}

\hint

Fjern parenteser:

\hint
\[
3x^2 + 2x + 5 + x^2 - 4x + 4
\]

\hint

Saml ens led sammen:

\hint
\[
(3x^2 + x^2) + (2x - 4x) + (5 + 4)
\]

\hint

Udregn hver gruppes sum:

\hint
\[
4x^2 - 2x + 9
\]

\end{exercise}

\begin{exercise}{Addition af polynomier 1-3}

Bestem $(x^3 + 2x^2 + x) + (2x^3 - x^2 + 3x)$.

\answerbox{3x^3 + x^2 + 4x}

\hint

Fjern parenteser og saml ens led:

\hint
\[
(x^3 + 2x^3) + (2x^2 - x^2) + (x + 3x)
\]

\hint

Udregn:

\hint
\[
3x^3 + x^2 + 4x
\]

\end{exercise}

\begin{exercise}{Addition af polynomier 1-5}

Bestem $(4x^2 + 3x - 1) + (2x^2 - x + 5)$.

\answerbox{6x^2 + 2x + 4}

\hint

Fjern parenteser og saml ens led:

\hint
\[
(4x^2 + 2x^2) + (3x - x) + (-1 + 5)
\]

\hint

Udregn:

\hint

6x^2 + 2x + 4
\]

\end{exercise}

\begin{exercise}{Addition af polynomier 1-6}

Bestem $(x^4 + 3x^3 + 2x) + (x^4 - 2x^3 + x)$.

\answerbox{2x^4 + x^3 + 3x}

\hint

Fjern parenteser og saml ens led:

\hint
\[
(x^4 + x^4) + (3x^3 - 2x^3) + (2x + x)
\]

\hint

Udregn:

\hint
\[
2x^4 + x^3 + 3x
\]

\end{exercise}
