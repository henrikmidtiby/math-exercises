\begin{exercise}{Subtraktion af polynomier 1-1}

Bestem $(5x + 3) - (2x - 4)$.

\answerbox{3x + 7}

\hint

Fjern parenteser og skift fortegn for det andet polynomium:

\hint
\[
5x + 3 - 2x + 4
\]

\hint

Saml ens led sammen:

\hint
\[
(5x - 2x) + (3 + 4)
\]

\hint

Udregn hver gruppes sum:

\hint
\[
3x + 7
\]

\end{exercise}

\begin{exercise}{Subtraktion af polynomier 1-6}

Bestem $(x^4 + 3x^3 + 2x) - (x^4 - 2x^3 + x)$.

\answerbox{5x^3 + x}

\hint

Fjern parenteser og skift fortegn:

\hint
\[
x^4 + 3x^3 + 2x - x^4 + 2x^3 - x
\]

\hint

Saml ens led:

\hint
\[
(x^4 - x^4) + (3x^3 + 2x^3) + (2x - x)
\]

\hint

Udregn:

\hint
\[
0 + 5x^3 + x = 5x^3 + x
\]

\end{exercise}
