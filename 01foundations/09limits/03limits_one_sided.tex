\begin{exercise}{Enside grænseværdier 1-1}

Bestem $\lim_{x \to 2^+} (x - 2)$.

\answerbox{0}

\hint

En højresidet grænse anvender værdier af $x$ som nærmer sig 2 fra højre.

\hint

Substituer $x$ tæt på 2 fra højre (f.eks. 2.1, 2.01, 2.001):

\hint
\[
\lim_{x \to 2^+} (x - 2) = 0
\]

\end{exercise}

\newpage

\begin{exercise}{Enside grænseværdier 1-2}

Bestem $\lim_{x \to 0^-} x$.

\answerbox{0}

\hint

En venstresidet grænse anvender værdier af $x$ som nærmer sig 0 fra venstre.

\hint

Substituer $x$ tæt på 0 fra venstre (f.eks. -0.1, -0.01, -0.001):

\hint
\[
\lim_{x \to 0^-} x = 0
\]

\end{exercise}

\begin{exercise}{Enside grænseværdier 1-3}

Bestem $\lim_{x \to 1^+} (x + 1)$.

\answerbox{2}

\hint

Substituer $x$ tæt på 1 fra højre:

\hint
\[
\lim_{x \to 1^+} (x + 1) = 1 + 1 = 2
\]

\end{exercise}

\newpage

\begin{exercise}{Enside grænseværdier 1-4}

Bestem $\lim_{x \to 3^-} (5 - x)$.

\answerbox{2}

\hint

Substituer $x$ tæt på 3 fra venstre:

\hint
\[
\lim_{x \to 3^-} (5 - x) = 5 - 3 = 2
\]

\end{exercise}

\begin{exercise}{Enside grænseværdier 1-5}

Bestem $\lim_{x \to 0^+} \frac{1}{x}$.

\answerbox{\infty}

\hint

Når $x$ nærmer sig 0 fra højre, bliver $\frac{1}{x}$ større og større.

\hint
\[
\lim_{x \to 0^+} \frac{1}{x} = \infty
\]

\end{exercise}

\newpage

\begin{exercise}{Enside grænseværdier 1-6}

Bestem $\lim_{x \to 0^-} \frac{1}{x}$.

\answerbox{-\infty}

\hint

Når $x$ nærmer sig 0 fra venstre, bliver $\frac{1}{x}$ negativt og større i absolut værdi.

\hint
\[
\lim_{x \to 0^-} \frac{1}{x} = -\infty
\]

\end{exercise}
