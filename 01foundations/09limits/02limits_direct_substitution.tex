\begin{exercise}{Direkte substitution 1-1}

Bestem $\lim_{x \to 2} (3x + 4)$.

\answerbox{10}

\hint

Substituer $x = 2$ direkt ind i funktionen.

\hint
\[
3(2) + 4 = 6 + 4 = 10
\]

\end{exercise}

\newpage

\begin{exercise}{Direkte substitution 1-2}

Bestem $\lim_{x \to -1} (x^2 + 2x - 3)$.

\answerbox{-4}

\hint

Substituer $x = -1$:

\hint
\[
(-1)^2 + 2(-1) - 3 = 1 - 2 - 3 = -4
\]

\end{exercise}

\newpage

\begin{exercise}{Direkte substitution 1-3}

Bestem $\lim_{x \to 3} \frac{x + 1}{x - 1}$.

\answerbox{2}

\hint

Substituer $x = 3$:

\hint
\[
\frac{3 + 1}{3 - 1} = \frac{4}{2} = 2
\]

\end{exercise}

\newpage

\begin{exercise}{Direkte substitution 1-4}

Bestem $\lim_{x \to 0} (5x^3 - 2x + 7)$.

\answerbox{7}

\hint

Substituer $x = 0$:

\hint
\[
5(0)^3 - 2(0) + 7 = 0 - 0 + 7 = 7
\]

\end{exercise}

\begin{exercise}{Direkte substitution 1-5}

Bestem $\lim_{x \to 4} \sqrt{x + 5}$.

\answerbox{3}

\hint

Substituer $x = 4$:

\hint
\[
\sqrt{4 + 5} = \sqrt{9} = 3
\]

\end{exercise}
