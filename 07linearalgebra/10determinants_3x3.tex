\begin{exercise}{Determinanter af 3x3 matricer 1-1}

Bestem determinanten af
\[
\begin{pmatrix}
1 & 2 & 3 \
0 & 1 & 4 \
5 & 6 & 0
\end{pmatrix}
\]

\answerbox{1}

\hint

Brug udvikling efter første række:
\[
\det(A) = 1 \cdot \det \begin{pmatrix} 1 & 4 \ 6 & 0 \end{pmatrix} - 2 \cdot \det \begin{pmatrix} 0 & 4 \ 5 & 0 \end{pmatrix} + 3 \cdot \det \begin{pmatrix} 0 & 1 \ 5 & 6 \end{pmatrix}
\]

\hint

Beregn første 2x2 determinant:
\[
\det \begin{pmatrix} 1 & 4 \ 6 & 0 \end{pmatrix} = (1)(0) - (4)(6) = -24
\]

\hint

Beregn andet 2x2 determinant:
\[
\det \begin{pmatrix} 0 & 4 \ 5 & 0 \end{pmatrix} = (0)(0) - (4)(5) = -20
\]

\hint

Beregn tredje 2x2 determinant:
\[
\det \begin{pmatrix} 0 & 1 \ 5 & 6 \end{pmatrix} = (0)(6) - (1)(5) = -5
\]

\hint

Sammt med fortegn:
\[
1\cdot(-24) - 2\cdot(-20) + 3\cdot(-5) = -24 + 40 - 15 = 1
\]

\end{exercise}

\begin{exercise}{Determinanter af 3x3 matricer 1-2}

Bestem determinanten af
\[
\begin{pmatrix}
2 & 0 & 1 \
0 & 3 & 0 \
1 & 0 & 4
\end{pmatrix}
\]

\answerbox{-21}

\hint

Brug udvikling efter anden række (indeholder to nuller):
\[
\det(A) = -3 \cdot \det \begin{pmatrix} 2 & 1 \ 1 & 4 \end{pmatrix}
\]

\hint

Beregn 2x2 determinant:
\[
\det \begin{pmatrix} 2 & 1 \ 1 & 4 \end{pmatrix} = (2)(4) - (1)(1) = 7
\]

\hint

Sammt med fortegn:
\[
\det(A) = -3 \cdot 7 = -21
\]

\end{exercise}

\begin{exercise}{Determinanter af 3x3 matricer 1-3}

Bestem determinanten af
\[
\begin{pmatrix}
1 & 0 & 0 \
0 & 2 & 0 \
0 & 0 & 3
\end{pmatrix}
\]

\answerbox{6}

\hint

Dette er en diagonalmatrix. Determinanten er produktet af diagonalelementerne.

\hint

Beregn:
\[
\det(A) = 1 \cdot 2 \cdot 3 = 6
\]

\end{exercise}

\begin{exercise}{Determinanter af 3x3 matricer 1-4}

Bestem determinanten af
\[
\begin{pmatrix}
1 & 1 & 1 \
0 & 1 & 1 \
0 & 0 & 1
\end{pmatrix}
\]

\answerbox{1}

\hint

Brug udvikling efter første kolonne:
\[
\det(A) = 1 \cdot \det \begin{pmatrix} 1 & 1 \ 0 & 1 \end{pmatrix}
\]

\hint

Beregn 2x2 determinant:
\[
\det \begin{pmatrix} 1 & 1 \ 0 & 1 \end{pmatrix} = 1
\]

\end{exercise}
