\begin{exercise}{Matrix addition 1-1}

Bestem
\[
\begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix} + \begin{pmatrix} 5 & 6 \\ 7 & 8 \end{pmatrix}
\]

\answerbox{\begin{pmatrix} 6 & 8 \\ 10 & 12 \end{pmatrix}}

\hint

Addition af matricer gøres ved at lægge tilsvarende elementer sammen.

\hint

Læg elementerne i første række sammen: $1 + 5 = 6$ og $2 + 6 = 8$

\hint

Læg elementerne i anden række sammen: $3 + 7 = 10$ og $4 + 8 = 12$

\hint

Resultatet er
\[
\begin{pmatrix} 6 & 8 \\ 10 & 12 \end{pmatrix}
\]

\end{exercise}

\begin{exercise}{Matrix addition 1-2}

Bestem
\[
\begin{pmatrix} 2 & 0 \\ -1 & 3 \end{pmatrix} + \begin{pmatrix} -2 & 4 \\ 1 & -3 \end{pmatrix}
\]

\answerbox{\begin{pmatrix} 0 & 4 \\ 0 & 0 \end{pmatrix}}

\hint

Læg tilsvarende elementer sammen:

\hint
\[
\begin{pmatrix} 2 + (-2) & 0 + 4 \\ -1 + 1 & 3 + (-3) \end{pmatrix} = \begin{pmatrix} 0 & 4 \\ 0 & 0 \end{pmatrix}
\]

\end{exercise}

\begin{exercise}{Matrix addition 1-3}

Bestem
\[
\begin{pmatrix} 3 & 1 \\ 2 & -1 \end{pmatrix} + \begin{pmatrix} 1 & 2 \\ -1 & 3 \end{pmatrix}
\]

\answerbox{\begin{pmatrix} 4 & 3 \\ 1 & 2 \end{pmatrix}}

\hint

Læg tilsvarende elementer sammen:

\hint
\[
\begin{pmatrix} 3 + 1 & 1 + 2 \\ 2 + (-1) & -1 + 3 \end{pmatrix} = \begin{pmatrix} 4 & 3 \\ 1 & 2 \end{pmatrix}
\]

\end{exercise}

\begin{exercise}{Matrix addition 1-4}

Bestem
\[
\begin{pmatrix} 0 & 1 & 2 \\ 3 & 4 & 5 \end{pmatrix} + \begin{pmatrix} 1 & 0 & -1 \\ -2 & -3 & -4 \end{pmatrix}
\]

\answerbox{\begin{pmatrix} 1 & 1 & 1 \\ 1 & 1 & 1 \end{pmatrix}}

\hint

Læg tilsvarende elementer sammen:

\hint
\[
\begin{pmatrix} 0 + 1 & 1 + 0 & 2 + (-1) \\ 3 + (-2) & 4 + (-3) & 5 + (-4) \end{pmatrix}
\]

\hint

Udregn resultaterne:

\hint
\[
\begin{pmatrix} 1 & 1 & 1 \\ 1 & 1 & 1 \end{pmatrix}
\]

\end{exercise}

\[

\]

\end{exercise}
