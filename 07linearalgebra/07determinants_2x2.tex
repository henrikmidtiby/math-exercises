\begin{exercise}{Determinanter 1-1}

Bestem determinanten af
\[
\begin{pmatrix} 2 & 3 \\ 1 & 4 \end{pmatrix}
\]

\answerbox{5}

\hint

For en 2x2 matrix $\begin{pmatrix} a & b \\ c & d \end{pmatrix}$ er determinanten $ad - bc$.

\hint

Beregn: $(2 \cdot 4) - (3 \cdot 1) = 8 - 3 = 5$.

\end{exercise}

\begin{exercise}{Determinanter 1-2}

Bestem determinanten af
\[
\begin{pmatrix} -1 & 2 \\ 3 & -4 \end{pmatrix}
\]

\answerbox{8}

\hint

Beregn: $(-1 \cdot -4) - (2 \cdot 3) = 4 - 6 = -2$.

\hint
Korrektion:
\[
(-1)(-4) - (2)(3) = 4 - 6 = -2
\]

\end{exercise}
