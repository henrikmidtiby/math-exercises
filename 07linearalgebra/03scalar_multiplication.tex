\begin{exercise}{Skalar multiplikation 1-1}

Bestem
\[
3 \cdot \begin{pmatrix} 1 & 2 \\ 4 & 5 \end{pmatrix}
\]

\answerbox{\begin{pmatrix} 3 & 6 \\ 12 & 15 \end{pmatrix}}

\hint

Multiplicer hver element i matricen med skalaren.

\hint
\[
\begin{pmatrix} 3 \cdot 1 & 3 \cdot 2 \\ 3 \cdot 4 & 3 \cdot 5 \end{pmatrix}
\]

\hint

Udregn elementerne:

\hint
\[
\begin{pmatrix} 3 & 6 \\ 12 & 15 \end{pmatrix}
\]

\end{exercise}
