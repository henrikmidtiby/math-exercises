\begin{exercise}{Lineære ligningssystemer 1-1}

Løs følgende ligningssystem:
\[
\begin{cases}
 x + y = 5 \
 x - y = 1
 \end{cases}
\]

\answerbox{\begin{pmatrix} x \ y \end{pmatrix} = \begin{pmatrix} 3 \ 2 \end{pmatrix}}

\hint

Addér de to ligninger:
\[
(x + y) + (x - y) = 5 + 1 \ 2x = 6
\]

\hint

Løs for x:
\[
x = 3
\]

\hint

Substituer i første ligning:
\[
3 + y = 5
\]

\hint

Løs for y:
\[
y = 2
\]

\end{exercise}

\begin{exercise}{Lineære ligningssystemer 1-2}

Løs følgende ligningssystem:
\[
\begin{cases}
 x + y = 5 \
 2x + 3y = 11
 \end{cases}
\]

\answerbox{\begin{pmatrix} x \ y \end{pmatrix} = \begin{pmatrix} 4 \ 1 \end{pmatrix}}

\hint

Løs første ligning for x:
\[
x = 5 - y
\]

\hint

Substituer i anden ligning:
\[
2(5 - y) + 3y = 11
\]

\hint

Udvik og forenk:
\[
10 - 2y + 3y = 11 \ 10 + y = 11
\]

\hint

Løs for y:
\[
y = 1
\]

\hint

Substituer tilbage:
\[
x = 5 - 1 = 4
\]

\end{exercise}

\begin{exercise}{Lineære ligningssystemer 1-3}

Løs følgende ligningssystem:
\[
\begin{cases}
 x + 2y = 8 \
 2x + y = 7
 \end{cases}
\]

\answerbox{\begin{pmatrix} x \ y \end{pmatrix} = \begin{pmatrix} 2 \ 3 \end{pmatrix}}

\hint

Multiplicér anden ligning med 2:
\[
4x + 2y = 14
\]

\hint

Træk første ligning fra:
\[
(4x + 2y) - (x + 2y) = 14 - 8 \ 3x = 6
\]

\hint

Løs for x:
\[
x = 2
\]

\hint

Substituer i første ligning:
\[
2 + 2y = 8
\]

\hint

Løs for y:
\[
2y = 6 \ y = 3
\]

\end{exercise}

\hint
